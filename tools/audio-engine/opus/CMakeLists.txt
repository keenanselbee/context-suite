cmake_minimum_required(VERSION 3.18)
project(ContextSuiteOpusDependency LANGUAGES C)

if(NOT WIN32 OR NOT CMAKE_C_COMPILER_ID STREQUAL "MSVC" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This dependency recipe requires Windows x64 MSVC.")
endif()
if(NOT IS_DIRECTORY "${OPUS_SOURCE}" OR NOT OPUS_PACKAGE_VERSION STREQUAL "1.6.1-contextsuite-g3da9f7a6db1c")
    message(FATAL_ERROR "Use the pinned source and version from Build-OpusDependency.ps1.")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING ON CACHE BOOL "" FORCE)
set(OPUS_DEEP_PLC OFF CACHE BOOL "" FORCE)
set(OPUS_DRED OFF CACHE BOOL "" FORCE)
set(OPUS_OSCE OFF CACHE BOOL "" FORCE)
set(OPUS_FIXED_POINT OFF CACHE BOOL "" FORCE)
set(OPUS_FLOAT_APPROX OFF CACHE BOOL "" FORCE)
set(OPUS_FAST_MATH OFF CACHE BOOL "" FORCE)
set(OPUS_HARDENING ON CACHE BOOL "" FORCE)
set(OPUS_STACK_PROTECTOR ON CACHE BOOL "" FORCE)
set(OPUS_STATIC_RUNTIME OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
enable_testing()
add_subdirectory("${OPUS_SOURCE}" opus)
target_compile_options(opus PRIVATE /GS /guard:cf /guard:ehcont)

# The pinned upstream revision sets MSVC_CLANG_CL to the list "false;BOOL".
# Correct the resulting per-source GCC flag for MSVC without editing its source.
get_target_property(opus_sources opus SOURCES)
set(corrected_sources 0)
foreach(source IN LISTS opus_sources)
    get_filename_component(absolute_source "${source}" ABSOLUTE BASE_DIR "${OPUS_SOURCE}")
    if(absolute_source MATCHES "/dnn/")
        message(FATAL_ERROR "Neural source entered the bounded Opus target.")
    endif()
    get_source_file_property(flags "${absolute_source}" TARGET_DIRECTORY opus COMPILE_FLAGS)
    if(flags STREQUAL "-msse4.1")
        set_source_files_properties("${absolute_source}" TARGET_DIRECTORY opus PROPERTIES COMPILE_FLAGS "")
        math(EXPR corrected_sources "${corrected_sources} + 1")
    endif()
endforeach()
if(corrected_sources EQUAL 0)
    message(FATAL_ERROR "Pinned upstream compiler workaround no longer matches; review the recipe.")
endif()
message(STATUS "Corrected MSVC source flags: ${corrected_sources}")

add_executable(ContextSuite.Opus.Version Version.c)
target_link_libraries(ContextSuite.Opus.Version PRIVATE Opus::opus)
target_compile_definitions(ContextSuite.Opus.Version PRIVATE EXPECTED_OPUS_VERSION="libopus ${OPUS_PACKAGE_VERSION}")
add_test(NAME ContextSuite.Opus.Version COMMAND ContextSuite.Opus.Version)
