cmake_minimum_required(VERSION 3.18)
project(ContextSuiteOggVorbisDependency LANGUAGES C)

if(NOT WIN32 OR NOT CMAKE_C_COMPILER_ID STREQUAL "MSVC" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This dependency recipe requires Windows x64 MSVC.")
endif()
if(NOT IS_DIRECTORY "${OGG_SOURCE}" OR NOT IS_DIRECTORY "${VORBIS_SOURCE}")
    message(FATAL_ERROR "Use the pinned sources from Build-AudioDependency.ps1.")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(INSTALL_DOCS OFF CACHE BOOL "" FORCE)
set(INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(INSTALL_CMAKE_PACKAGE_MODULE OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
enable_testing()
add_subdirectory("${OGG_SOURCE}" ogg)
add_subdirectory("${VORBIS_SOURCE}" vorbis)
