cmake_minimum_required(VERSION 3.24)
project(ContextSuiteOfficeProcessProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The process probe requires Windows x64.")
endif()
add_executable(ContextSuite.Office.ProcessProbe Probe.cpp)
target_compile_features(ContextSuite.Office.ProcessProbe PRIVATE cxx_std_20)
target_compile_options(ContextSuite.Office.ProcessProbe PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(ContextSuite.Office.ProcessProbe PRIVATE /Brepro)
target_link_libraries(ContextSuite.Office.ProcessProbe PRIVATE advapi32)
set_target_properties(ContextSuite.Office.ProcessProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
