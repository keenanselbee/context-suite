cmake_minimum_required(VERSION 3.24)
project(ContextSuitePdfiumProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The evaluation probe requires Windows x64.")
endif()
if(NOT EXISTS "${PDFIUM_PAYLOAD}/include/fpdfview.h" OR NOT EXISTS "${PDFIUM_PAYLOAD}/lib/pdfium.dll.lib")
  message(FATAL_ERROR "Supply the verified PDFium evaluation payload.")
endif()
add_executable(ContextSuite.Pdfium.Probe Probe.cpp)
target_compile_features(ContextSuite.Pdfium.Probe PRIVATE cxx_std_20)
target_compile_options(ContextSuite.Pdfium.Probe PRIVATE /W4 /WX /utf-8)
target_include_directories(ContextSuite.Pdfium.Probe SYSTEM PRIVATE "${PDFIUM_PAYLOAD}/include")
target_link_libraries(ContextSuite.Pdfium.Probe PRIVATE "${PDFIUM_PAYLOAD}/lib/pdfium.dll.lib")
set_target_properties(ContextSuite.Pdfium.Probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_command(TARGET ContextSuite.Pdfium.Probe POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFIUM_PAYLOAD}/bin/pdfium.dll" "$<TARGET_FILE_DIR:ContextSuite.Pdfium.Probe>")
