cmake_minimum_required(VERSION 3.21)
project(ContextSuiteDds LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Context Suite DDS requires Windows x64")
endif()
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_DX11 OFF CACHE BOOL "" FORCE)
set(BUILD_DX12 OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BC_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENEXR_SUPPORT OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBJPEG_SUPPORT OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBPNG_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${DIRECTXTEX_SOURCE}" directxtex)
add_library(ContextSuite.Dds.Native SHARED "${CMAKE_CURRENT_LIST_DIR}/../../proprietary/native/dds/Bridge.cpp")
target_compile_features(ContextSuite.Dds.Native PRIVATE cxx_std_17)
target_compile_definitions(ContextSuite.Dds.Native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ContextSuite.Dds.Native PRIVATE /W4 /WX /permissive-)
target_link_libraries(ContextSuite.Dds.Native PRIVATE DirectXTex)
target_link_options(ContextSuite.Dds.Native PRIVATE "/MAP:${CMAKE_BINARY_DIR}/ContextSuite.Dds.Native.map")
set_target_properties(ContextSuite.Dds.Native PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
