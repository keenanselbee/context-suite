cmake_minimum_required(VERSION 3.24)
project(ContextSuitePdfRenderer LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The PDF renderer requires Windows x64.")
endif()
if(NOT EXISTS "${PDFIUM_PAYLOAD}/include/fpdfview.h" OR NOT EXISTS "${PDFIUM_PAYLOAD}/lib/pdfium.dll.lib")
  message(FATAL_ERROR "Supply the verified non-V8 PDFium payload.")
endif()
if(NOT EXISTS "${QPDF_PAYLOAD}/include/qpdf/QPDF.hh" OR NOT EXISTS "${QPDF_PAYLOAD}/lib/qpdf.lib")
  message(FATAL_ERROR "Supply the verified qpdf x64 evaluation SDK for page geometry.")
endif()
add_executable(ContextSuite.PdfRenderer Renderer.cpp)
target_compile_features(ContextSuite.PdfRenderer PRIVATE cxx_std_20)
target_compile_options(ContextSuite.PdfRenderer PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(ContextSuite.PdfRenderer PRIVATE /Brepro)
target_include_directories(ContextSuite.PdfRenderer SYSTEM PRIVATE "${PDFIUM_PAYLOAD}/include")
target_include_directories(ContextSuite.PdfRenderer SYSTEM PRIVATE "${QPDF_PAYLOAD}/include")
target_link_libraries(ContextSuite.PdfRenderer PRIVATE "${PDFIUM_PAYLOAD}/lib/pdfium.dll.lib" "${QPDF_PAYLOAD}/lib/qpdf.lib")
set_target_properties(ContextSuite.PdfRenderer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_command(TARGET ContextSuite.PdfRenderer POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFIUM_PAYLOAD}/bin/pdfium.dll" "$<TARGET_FILE_DIR:ContextSuite.PdfRenderer>")
foreach(runtime qpdf30.dll concrt140.dll msvcp140.dll msvcp140_1.dll msvcp140_2.dll
    msvcp140_atomic_wait.dll msvcp140_codecvt_ids.dll vcruntime140.dll vcruntime140_1.dll)
  add_custom_command(TARGET ContextSuite.PdfRenderer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${QPDF_PAYLOAD}/bin/${runtime}" "$<TARGET_FILE_DIR:ContextSuite.PdfRenderer>")
endforeach()
