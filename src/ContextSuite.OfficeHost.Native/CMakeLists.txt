cmake_minimum_required(VERSION 3.24)
project(ContextSuiteOfficeHost LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Office host requires Windows x64.")
endif()
add_executable(ContextSuite.OfficeHost Host.cpp)
target_compile_features(ContextSuite.OfficeHost PRIVATE cxx_std_20)
target_compile_options(ContextSuite.OfficeHost PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(ContextSuite.OfficeHost PRIVATE /Brepro "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
target_link_libraries(ContextSuite.OfficeHost PRIVATE bcrypt userenv advapi32 user32)
set_target_properties(ContextSuite.OfficeHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
