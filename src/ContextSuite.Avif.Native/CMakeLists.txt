cmake_minimum_required(VERSION 3.24)
project(ContextSuiteAvifNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The AVIF bridge requires Windows x64.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT DEFINED AVIF_SOURCE_DIR OR NOT DEFINED AVIF_LIBRARY OR NOT DEFINED AOM_LIBRARY)
  message(FATAL_ERROR "Pass pinned AVIF_SOURCE_DIR, AVIF_LIBRARY and AOM_LIBRARY.")
endif()
add_library(ContextSuite.Avif.Native SHARED Encoder.cpp)
target_compile_features(ContextSuite.Avif.Native PRIVATE cxx_std_20)
target_compile_options(ContextSuite.Avif.Native PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(ContextSuite.Avif.Native PRIVATE /Brepro)
target_include_directories(ContextSuite.Avif.Native PRIVATE "${AVIF_SOURCE_DIR}/include")
target_link_libraries(ContextSuite.Avif.Native PRIVATE "${AVIF_LIBRARY}" "${AOM_LIBRARY}")
