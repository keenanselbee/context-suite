cmake_minimum_required(VERSION 3.24)
project(ContextSuiteHeifNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The HEIF bridge requires Windows x64.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(libheif 1.23.4 EXACT CONFIG REQUIRED)
add_library(ContextSuite.Heif.Native SHARED Decoder.cpp)
target_compile_features(ContextSuite.Heif.Native PRIVATE cxx_std_20)
target_compile_options(ContextSuite.Heif.Native PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(ContextSuite.Heif.Native PRIVATE /Brepro)
target_link_libraries(ContextSuite.Heif.Native PRIVATE heif)
