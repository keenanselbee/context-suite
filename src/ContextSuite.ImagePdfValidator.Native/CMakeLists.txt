cmake_minimum_required(VERSION 3.24)
project(ContextSuiteImagePdfValidator LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The image PDF validator requires Windows x64.")
endif()
if(NOT EXISTS "${QPDF_PAYLOAD}/include/qpdf/QPDF.hh" OR NOT EXISTS "${QPDF_PAYLOAD}/lib/qpdf.lib")
  message(FATAL_ERROR "Supply the verified qpdf x64 evaluation SDK.")
endif()
add_executable(ContextSuite.ImagePdfValidator Validator.cpp)
target_compile_features(ContextSuite.ImagePdfValidator PRIVATE cxx_std_20)
target_compile_options(ContextSuite.ImagePdfValidator PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(ContextSuite.ImagePdfValidator PRIVATE /Brepro)
target_include_directories(ContextSuite.ImagePdfValidator SYSTEM PRIVATE "${QPDF_PAYLOAD}/include")
target_link_libraries(ContextSuite.ImagePdfValidator PRIVATE "${QPDF_PAYLOAD}/lib/qpdf.lib" bcrypt)
set_target_properties(ContextSuite.ImagePdfValidator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(runtime qpdf30.dll concrt140.dll msvcp140.dll msvcp140_1.dll msvcp140_2.dll
    msvcp140_atomic_wait.dll msvcp140_codecvt_ids.dll vcruntime140.dll vcruntime140_1.dll)
  add_custom_command(TARGET ContextSuite.ImagePdfValidator POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${QPDF_PAYLOAD}/bin/${runtime}" "$<TARGET_FILE_DIR:ContextSuite.ImagePdfValidator>")
endforeach()
